python3 -c "
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Image
from cv_bridge import CvBridge
from ultralytics import YOLO
from pathlib import Path

model = YOLO('/workspaces/mirte_ws/src/mirte_detectio/mirte_detectio/best.pt')
print('Model loaded, classes:', model.names)

rclpy.init()
node = Node('test_yolo')
bridge = CvBridge()

def cb(msg):
    img = bridge.imgmsg_to_cv2(msg, 'bgr8')
    results = model(img)
    for r in results:
        print('Detections:', len(r.boxes))
        for i in range(len(r.boxes)):
            print(f'  class={r.names[int(r.boxes.cls[i])]}, conf={r.boxes.conf[i]:.2f}')
    rclpy.shutdown()

node.create_subscription(Image, '/camera/color/image_raw', cb, 10)
print('Waiting for one frame...')
rclpy.spin(node)
"
